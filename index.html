📖 내용

개요

이 프로젝트는 MZ세대 대상으로 한 맞춤형 향수 추천 쇼핑몰 입니다.
이 프로젝트의 목표는 개성과 취향을 중시하는 MZ 세대가
자신의 스타일과 성격에 맞는 향수를 쉽게 찾을 수 있도록 돕는 것입니다.

프로젝트 목표

▶ 맞춤형 향수 추천을 통해 사용자의 선택의 어려움을 줄이기
▶ 향수 입문 자들이 쉽게 접근할 수 있는 환경 조성
▶ MZ 세대를 타겟 으로 한 효과적인 마케팅 전략 수립

주요 기능

▶ 옷 스타일에 맞는 향수 추천 테스트
  ▪ 사용자가 원하는 패션 스타일을 3가지 선택하면
해당 스타일에 어울리는 향수를 추천
▶ 향수 MBTI 테스트
  ▪ MBTI 성격 유형과 유사한 테스트를 통해 사용자의 성격과
    취향을 분석, 맞춤형 향수 추천

💠토큰 교환 로직

useEffect(() => {
    const exchangeToken = async () => {
        try {
            const response = await fetch("https://fit-conference.shop/api/v1/auth/token-exchange", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
              credentials: "include",
            });
            
            if (response.ok) {
              let token = response.headers.get("Authorization");
              if (token?.startsWith("Bearer ")) token = token.substring(7);
              localStorage.setItem("access-token", token);
            
              const userInfo = await axios.get("https://fit-conference.shop/api/v1/users/account", {
                headers: { Authorization: `Bearer ${token}` },
              });
            
              dispatch(loginSuccess({ user: userInfo.data.response, token }));
              navigate("/mypage");
            }
          } catch (error) {
            console.error("Token exchange failed:", error);
          }
        };
            
        exchangeToken();
      }, []);

    auth-slice.js
    const initialState = {
  job: null,
  experience: null,
  interests: [],
};
            
const authSlice = createSlice({
  name: 'auth',
  initialState: initialState,
  reducers: {
    updateProfile(state, action) {
      state.name = action.payload.name;
      state.job = action.payload.job;
      state.experience = action.payload.experience;
      state.interests = action.payload.interests;
    },
  },
});
            
export const {
  updateProfile,
} = authSlice.actions;
            
export default authSlice.reducer;

      'https://fit-conference.shop/api/v1/users/profile/image',
      formData,

          'Content-Type': 'multipart/form-data',
        },

    );
